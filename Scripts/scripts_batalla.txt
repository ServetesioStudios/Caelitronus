func _on_carta_soltada(carta: CardData, posicion_global: Vector2) -> void:

	var objetivo := _resolver_objetivo(
		carta,
		posicion_global
	)

	if objetivo == null:
		return

	intentar_jugar_carta(
		carta,
		objetivo
	)


func _resolver_objetivo(
	carta: CardData,
	posicion_global: Vector2
) -> CombatEntity:

	match carta.tipo_objetivo:

		CardData.TipoObjetivo.PROPIO:
			return player


		CardData.TipoObjetivo.ENEMIGO_FIJO:
			return enemy


		CardData.TipoObjetivo.ENEMIGO_CUALQUIERA:
			return _entidad_mas_cercana(
				posicion_global,
				[enemy]
			)


		CardData.TipoObjetivo.LIBRE:
			return _entidad_mas_cercana(
				posicion_global,
				[player, enemy]
			)


	return null


func _entidad_mas_cercana(
	posicion_global: Vector2,
	candidatos: Array
) -> CombatEntity:

	var mejor: CombatEntity = null
	var mejor_dist := INF

	for c in candidatos:

		if not is_instance_valid(c):
			continue

		var dist = c.global_position.distance_to(
			posicion_global
		)

		if dist < mejor_dist:
			mejor_dist = dist
			mejor = c

	return mejor

	# --------------------------------------------------------

func mostrar_texto(texto: String) -> void:
	if texto_combate == null:
		return

	historial_texto += texto + "\n\n"

	var lineas = historial_texto.split("\n")

	if lineas.size() > max_lineas:

		var nuevas_lineas = []

		for i in range(
			lineas.size() - max_lineas,
			lineas.size()
		):
			nuevas_lineas.append(
				lineas[i]
			)

		historial_texto = "\n".join(
			nuevas_lineas
		)


	texto_combate.text = historial_texto

	await get_tree().process_frame

	if is_instance_valid(texto_combate):
		texto_combate.scroll_to_line(
			texto_combate.get_line_count()
		)

		if not is_inside_tree() or estado != Estado.TURNO_ENEMIGO:

class_name EtherManager
extends Node

signal eteres_actualizados

enum ColorEter {
	ROJO,

const CANTIDAD_ETERES := 5
const REPETICIONES_POR_COMBATE := 2

var perfil: EtherProfileData
var eteres: Array[int] = []
var repeticiones_restantes: int = 0

var rng := RandomNumberGenerator.new()


func _ready() -> void:
	rng.randomize()


func iniciar_combate(nuevo_perfil: EtherProfileData) -> void:
	if nuevo_perfil == null:
		push_error("EtherManager: falta el perfil de Ã©ter.")
		return

	perfil = nuevo_perfil
	eteres.clear()
	repeticiones_restantes = REPETICIONES_POR_COMBATE

	repeticiones_actualizadas.emit(repeticiones_restantes)
	eteres_actualizados.emit()


func iniciar_turno() -> void:
	if perfil == null:
		push_error("EtherManager: no hay perfil configurado.")
		return

	eteres.clear()

	for i in range(CANTIDAD_ETERES):
		eteres.append(_generar_color())

	eteres_actualizados.emit()


func _generar_color() -> int:
	var pesos: Array[int] = [
		perfil.rojo,
		perfil.violeta,
		perfil.verde,
		perfil.celeste,
		perfil.rosa
	]

	var total := 0

	for peso in pesos:
		total += max(peso, 0)

	if total <= 0:
		push_error("EtherManager: todos los pesos son cero.")
		return -1

	var resultado := rng.randi_range(1, total)
	var acumulado := 0

	for i in range(pesos.size()):
		acumulado += max(pesos[i], 0)

		if resultado <= acumulado:
			return i

	return -1


func repetir_eteres(indices: Array[int]) -> bool:
	if repeticiones_restantes <= 0:
		return false

	if indices.is_empty():
		return false

	if perfil == null:
		return false

	var indices_unicos: Dictionary = {}

	for indice in indices:
		if indice < 0 or indice >= eteres.size():
			return false

		if eteres[indice] == -1:
			return false

		indices_unicos[indice] = true

	for indice in indices_unicos:
		eteres[indice] = _generar_color()

	repeticiones_restantes -= 1

	eteres_actualizados.emit()
	repeticiones_actualizadas.emit(repeticiones_restantes)

	return true


func finalizar_turno() -> void:
	eteres.clear()
	eteres_actualizados.emit()

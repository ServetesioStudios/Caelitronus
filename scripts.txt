@export var sprite: AnimatedSprite2D

var animacion_idle: String = ""


func _ready() -> void:
	if sprite == null or sprite.sprite_frames == null:
		push_warning("AnimadorPersonaje: SpriteFrames no configurado.")
		return

	configurar_idle()
	reproducir_idle()


func configurar_idle() -> void:
	var frames := sprite.sprite_frames

	if frames.has_animation("idle"):
		animacion_idle = "idle"
		return

	if GameManager.player_data == null:
		push_warning("AnimadorPersonaje: no hay PlayerData para determinar el idle.")
		return

	match GameManager.player_data.tipo_caelius:
		GameManager.TipoCaelius.PENA:
			animacion_idle = "idle_pena"
		GameManager.TipoCaelius.IRA:
			animacion_idle = "idle_ira"
		GameManager.TipoCaelius.EGO:
			animacion_idle = "idle_ego"

	if not frames.has_animation(animacion_idle):
		push_warning("AnimadorPersonaje: no existe la animaciÃ³n %s." % animacion_idle)
		animacion_idle = ""


func reproducir_idle() -> void:
	if not is_instance_valid(sprite):
		return

	if sprite.sprite_frames == null or animacion_idle.is_empty():
		return

	if not sprite.sprite_frames.has_animation(animacion_idle):
		return

	sprite.play(animacion_idle)


func reproducir(nombre: String) -> void:
	if not is_instance_valid(sprite) or sprite.sprite_frames == null:
		push_warning("AnimadorPersonaje: sprite no configurado.")
		return

	if not sprite.sprite_frames.has_animation(nombre):
		push_warning("AnimadorPersonaje: animaciÃ³n '%s' no encontrada." % nombre)
		return

	sprite.play(nombre)

	if sprite.sprite_frames.get_animation_loop(nombre):
		push_warning("AnimadorPersonaje: la animaciÃ³n '%s' estÃ¡ configurada en bucle." % nombre)
		return

	await sprite.animation_finished

	if not is_instance_valid(sprite):
		return

	if sprite.animation != nombre:
		return

	animacion_terminada.emit(nombre)
	reproducir_idle()

func _resolver_objetivo(
	carta: CardData,
	posicion_global: Vector2
) -> CombatEntity:

	match carta.tipo_objetivo:

		CardData.TipoObjetivo.PROPIO:
			return player


		CardData.TipoObjetivo.ENEMIGO_FIJO:
			return enemy


		CardData.TipoObjetivo.ENEMIGO_CUALQUIERA:
			return _entidad_mas_cercana(
				posicion_global,
				[enemy]
			)


		CardData.TipoObjetivo.LIBRE:
			return _entidad_mas_cercana(
				posicion_global,
				[player, enemy]
			)


	return null


func _entidad_mas_cercana(
	posicion_global: Vector2,
	candidatos: Array
) -> CombatEntity:

	var mejor: CombatEntity = null
	var mejor_dist := INF

	for c in candidatos:

		if not is_instance_valid(c):
			continue

		var dist = c.global_position.distance_to(
			posicion_global
		)

		if dist < mejor_dist:
			mejor_dist = dist
			mejor = c

	return mejor

func generar_area(area: AreaAparicion) -> void:
	var colision := area.get_node_or_null(
        "CollisionShape2D"
	) as CollisionShape2D

	if colision == null or not colision.shape is RectangleShape2D:
		push_warning("Ãrea sin rectÃ¡ngulo vÃ¡lido: %s" % area.area_id)
		return

	var rectangulo := colision.shape as RectangleShape2D
	var mitad := rectangulo.size / 2.0
	var limite := mitad - Vector2(MARGEN, MARGEN)

	if limite.x <= 0.0 or limite.y <= 0.0:
		push_warning("Ãrea demasiado pequeÃ±a: %s" % area.area_id)
		return

	var posiciones: Array[Vector2] = []
	var indice_enemigo: int = 0

	for configuracion in area.enemigos:
		if configuracion == null or configuracion.enemy_data == null:
			continue

		for i in range(configuracion.cantidad):
			var encontrada := false
			var posicion := Vector2.ZERO

			for intento in range(MAX_INTENTOS):
				var posicion_local := Vector2(
					randf_range(-limite.x, limite.x),
					randf_range(-limite.y, limite.y)
				)

				posicion = colision.to_global(posicion_local)

				var separacion_correcta := true

				for anterior in posiciones:
					if posicion.distance_to(anterior) < area.separacion_minima:
						separacion_correcta = false
						break

				if separacion_correcta and posicion_libre_de_obstaculos(posicion):
					encontrada = true
					break

			if not encontrada:
				push_warning(
					"No se pudo generar %s en %s" % [
						configuracion.enemy_data.nombre,
						area.area_id
					]
				)
				continue

			posiciones.append(posicion)


			var enemigo_id := "%s_%d" % [area.area_id, indice_enemigo]
			
			crear_enemigo_mundo(
				configuracion.enemy_data,
				posicion, 
				enemigo_id
			)
			
			enemigos_generados.append({
				"id": enemigo_id,
				"enemy_data": configuracion.enemy_data,
				"posicion": posicion,
				"derrotado": false
			})

			indice_enemigo += 1


			print(
				"[%s] %s generado en %s" % [
					area.area_id,
					configuracion.enemy_data.nombre,
					posicion
				]
			)


func posicion_libre_de_obstaculos(posicion: Vector2) -> bool:
	var forma := RectangleShape2D.new()
	forma.size = TAMANO_MARCADOR

	var consulta := PhysicsShapeQueryParameters2D.new()
	consulta.shape = forma
	consulta.transform = Transform2D(0.0, posicion)
	consulta.collision_mask = CAPA_OBSTACULOS
	consulta.collide_with_bodies = true #StaticBody2D
	consulta.collide_with_areas = false #Area2D
	
	var espacio := areas_aparicion.get_world_2d().direct_space_state
	var resultados := espacio.intersect_shape(consulta, 1)
	
	return resultados.is_empty()

func _actualizar_disponibilidad(card_ui: CardUI) -> void:
	if card_ui.carta == null:
		print("Carta nula encontrada en:", card_ui.name)
		return

	var carta: CardData = card_ui.carta

	var energia_insuficiente: bool = not player_ref.puede_pagar(carta.costo)

	var es_carta_energia: bool = (
		carta.efectos.size() == 1
		and carta.efectos[0] is EnergyEffectData
	)

	var energia_completa: bool = player_ref.energia_actual >= player_ref.energia_max

	var carta_bloqueada: bool = energia_insuficiente or (
		es_carta_energia and energia_completa
	)

	if carta_bloqueada:
		card_ui.modulate = Color(0.5, 0.5, 0.5, 0.6)
	else:
		card_ui.modulate = Color(1, 1, 1, 1)
